function drawBoardLines(){
  const wrap = document.getElementById('boardWrap');
  const svg = document.getElementById('boardSvg');
  if(!wrap || !svg) return;
  const pairs = [['m1','e1'],['e1','a1'],['m1','e4'],['e1','m2'],['m2','m3'],['e4','m3']];
  const rect = wrap.getBoundingClientRect();
  svg.setAttribute('viewBox', `0 0 ${rect.width} ${rect.height}`);
  svg.innerHTML = pairs.map(([a,b])=>{
    const na = wrap.querySelector(`[data-node="${a}"]`), nb = wrap.querySelector(`[data-node="${b}"]`);
    if(!na||!nb) return '';
    const ra = na.getBoundingClientRect(), rb = nb.getBoundingClientRect();
    const x1 = ra.left-rect.left+ra.width/2, y1 = ra.top-rect.top+ra.height/2;
    const x2 = rb.left-rect.left+rb.width/2, y2 = rb.top-rect.top+rb.height/2;
    return `<line x1="${x1}" y1="${y1}" x2="${x2}" y2="${y2}"/>`;
  }).join('');
}

function renderSearchResults(q){
  q = q.trim().toLowerCase();
  const filtered = q ? SEARCH_INDEX.filter(i => i.name.toLowerCase().includes(q) || i.desc.toLowerCase().includes(q)) : SEARCH_INDEX.slice(0,8);
  if(filtered.length===0){
    searchResults.innerHTML = `<div class="search-empty">NO MATCHING RECORDS FOUND</div>`;
    return;
  }
  const groups = {};
  filtered.forEach(i => { (groups[i.cat] = groups[i.cat]||[]).push(i); });
  searchResults.innerHTML = Object.entries(groups).map(([cat, items])=>`
    <div class="sr-group">
      <div class="sr-group-label">${cat}</div>
      ${items.map(i=>`<div class="sr-item" data-name="${i.name}"><div><div class="n">${i.name}</div><div class="d">${i.desc}</div></div><div class="c">${cat}</div></div>`).join('')}
    </div>`).join('');
  searchResults.querySelectorAll('.sr-item').forEach((el, idx)=>{
    el.addEventListener('click', ()=>{
      const item = filtered[idx];
      closeSearch();
      item.open();
    });
  });
}
searchInput.addEventListener('input', (e)=>renderSearchResults(e.target.value));

document.addEventListener('keydown', (e)=>{
  if((e.metaKey || e.ctrlKey) && e.key.toLowerCase()==='k'){
    e.preventDefault();
    searchOverlay.classList.contains('open') ? closeSearch() : openSearch();
  }
  if(e.key==='Escape'){
    if(searchOverlay.classList.contains('open')) closeSearch();
    if(scrim.classList.contains('open')) closeModal();
  }
});

/* ============================================================
   FINGERPRINT ACCESS SEQUENCE
   ============================================================ */
const scanner = document.getElementById('scanner');
const scannerBtn = document.getElementById('scannerBtn');
const ringFg = document.getElementById('ringFg');
const accessStatus = document.getElementById('accessStatus');
const accessHint = document.getElementById('accessHint');
const particleHost = document.getElementById('particleHost');
const CIRC = 2 * Math.PI * 80;
ringFg.style.strokeDasharray = CIRC;

let holding = false, progress = 0, raf = null, lastTs = null;
const DURATION = 1650; // ms to full scan

function spawnParticles(){
  particleHost.innerHTML = '';
  for(let i=0;i<10;i++){
    const p = document.createElement('span');
    p.className='particle';
    const angle = Math.random()*Math.PI*2;
    const dist = 40 + Math.random()*30;
    p.style.setProperty('--px', Math.cos(angle)*dist+'px');
    p.style.setProperty('--py', Math.sin(angle)*dist+'px');
    p.style.left='50%'; p.style.top='50%';
    p.style.animationDelay = (Math.random()*1.2)+'s';
    particleHost.appendChild(p);
  }
}

function statusForProgress(p){
  if(p < 35) return 'VERIFYING IDENTITY…';
  if(p < 75) return 'ANALYZING BIOMETRIC SIGNATURE…';
  if(p < 100) return 'ACCESSING ÆZ NETWORK…';
  return 'IDENTITY VERIFIED';
}

function tick(ts){
  if(!holding) return;
  if(lastTs==null) lastTs = ts;
  const dt = ts - lastTs; lastTs = ts;
  progress = Math.min(100, progress + (dt/DURATION)*100);
  const offset = CIRC * (1 - progress/100);
  ringFg.style.strokeDashoffset = offset;
  accessStatus.innerHTML = `${statusForProgress(progress)} <span class="pct">${Math.floor(progress)}%</span>`;
  if(progress >= 100){
    finishScan();
    return;
  }
  raf = requestAnimationFrame(tick);
}

function startHold(){
  if(holding) return;
  holding = true; lastTs = null;
  scanner.classList.add('active','pressed');
  accessHint.textContent = 'HOLD STEADY…';
  spawnParticles();
  raf = requestAnimationFrame(tick);
}
function cancelHold(){
  if(!holding) return;
  holding = false;
  cancelAnimationFrame(raf);
  scanner.classList.remove('pressed');
  if(progress < 100){
    scanner.classList.remove('active');
    accessStatus.textContent='';
    accessHint.textContent = 'SCAN INTERRUPTED — HOLD TO RETRY';
    progress = 0;
    ringFg.style.strokeDashoffset = CIRC;
  }
}
function finishScan(){
  holding = false;
  cancelAnimationFrame(raf);
  accessHint.textContent='';
  setTimeout(enterArchive, 600);
}

scannerBtn.addEventListener('pointerdown', (e)=>{ e.preventDefault(); startHold(); });
window.addEventListener('pointerup', cancelHold);
scannerBtn.addEventListener('pointercancel', cancelHold);
scannerBtn.addEventListener('contextmenu', (e)=>e.preventDefault());
// keyboard accessibility: space/enter to hold isn't natural for long-press; provide instant fallback instead
scannerBtn.setAttribute('tabindex','0');
scannerBtn.setAttribute('role','button');
scannerBtn.setAttribute('aria-label','Press and hold to simulate identity verification, or use Enter Archive button below');

document.getElementById('fallbackBtn').addEventListener('click', enterArchive);

const BOOT_LINES = [
  'ESTABLISHING SECURE CONNECTION',
  'DECRYPTING ARCHIVE INDEX',
  'WELCOME, OPERATIVE'
];
function enterArchive(){
  const access = document.getElementById('access');
  const bootEl = document.getElementById('bootLines');
  let i = 0;
  bootEl.innerHTML = '';
  const iv = setInterval(()=>{
    if(i < BOOT_LINES.length){
      const line = document.createElement('div');
      line.textContent = BOOT_LINES[i];
      bootEl.appendChild(line);
      i++;
    } else {
      clearInterval(iv);
      access.classList.add('leaving');
      setTimeout(()=>{
        access.style.display='none';
        document.getElementById('app').classList.add('visible');
      }, 850);
    }
  }, 380);
}
